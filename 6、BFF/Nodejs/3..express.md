fetch和axios是发起网络请求，而express则是在nodejs后端接收和处理网络请求。我们首先看看使用原生nodejs响应服务请求。

## 1. 原生 Node.js HTTP 处理
原生http处理使用`createService`来创建 HTTP 服务器（官网：https://www.nodeapp.cn/http.html#http_class_http_agent）
```js
// Node.js 自带了一些核心模块，比如 `http`、`url`、`fs`、`path`等，可以直接通过 `require` 引入
const http = require('http');
const url = require('url');

// 创建 HTTP 服务器
const server = http.createServer((req, res) => {
  const { pathname, query } = url.parse(req.url, true);
  
  // 1. 处理 GET 请求
  if (req.method === 'GET') {
    if (pathname === '/') {
      res.writeHead(200, { 'Content-Type': 'text/plain' });
      res.end('Home Page'); // 每个响应都必须调用 res.end()，否则客户端会一直等待直到超时
    } 
    else if (pathname === '/about') {
      // 处理查询参数
      const name = query.name || 'Guest';
      res.writeHead(200, { 'Content-Type': 'application/json' });
      res.end(JSON.stringify({ message: `Hello ${name}` }));
    }
    else {
      res.writeHead(404);
      res.end('Not Found');
    }
  }
  
  // 2. 处理 POST 请求
  else if (req.method === 'POST') {
    if (pathname === '/echo') {
      let body = '';
      
      // 手动收集请求体数据流
      req.on('data', chunk => {
        body += chunk.toString();
      });
      
      // 数据接收完毕
      req.on('end', () => {
        try {
          // 尝试解析 JSON
          const data = JSON.parse(body);
          res.writeHead(200, { 'Content-Type': 'application/json' });
          res.end(JSON.stringify({ 
            received: true, 
            data 
          }));
        } catch (error) {
          res.writeHead(400);
          res.end('Invalid JSON');
        }
      });
    } else {
      res.writeHead(404);
      res.end('Not Found');
    }
  }
  
  // 3. 处理其他 HTTP 方法
  else {
    res.writeHead(405);
    res.end('Method Not Allowed');
  }
});

// 启动服务器
server.listen(3000, () => {
  console.log('Server running on http://localhost:3000');
});
```

## 2、Express.js 实现（使用框架）

### 2.1 基础用法

```js
const express = require('express');
const app = express();

// 中间件：自动解析 JSON 请求体
app.use(express.json());

// 1. 处理 GET 请求
app.get('/', (req, res) => {
  res.send('Home Page');
});

app.get('/about', (req, res) => {
  // 自动处理查询参数
  const name = req.query.name || 'Guest';
  res.json({ message: `Hello ${name}` });
});

// 2. 处理 POST 请求
app.post('/echo', (req, res) => {
  // 自动解析的请求体数据
  res.json({ 
    received: true, 
    data: req.body 
  });
});

// 3. 处理未匹配路由 (404)
app.use((req, res) => {
  res.status(404).send('Not Found');
});

// 4. 统一错误处理
app.use((err, req, res, next) => {
  console.error(err.stack);
  res.status(500).send('Server Error');
});

// 启动服务器
app.listen(3000, () => {
  console.log('Server running on http://localhost:3000');
});
```

### 2.2 路由参数（`req.params`）

除了查询参数 `req.query`，RESTful API 最常用的是**路径参数**，比如 `GET /users/123` 中的 `123`。

```js
// :id 是占位符，匹配到的值可以通过 req.params.id 获取
app.get('/users/:id', (req, res) => {
  console.log(req.params.id);  // 如果请求 /users/123，输出 '123'
  // 根据 id 查找用户...
  res.json({ id: req.params.id, name: 'John' });
});

// 可以有多个参数
app.get('/posts/:year/:month', (req, res) => {
  console.log(req.params.year, req.params.month); // 2024, 06
});
```

**`req.query` 与 `req.params` 的区分**：
- `/users?name=john` → `req.query.name` → 查询字符串参数
- `/users/123` → `req.params.id` → 路径参数，是 URL 的一部分

### 2.3 路由模块化（`express.Router()`）

真实项目有几十上百个接口，全部写在一个文件里无法维护。用 `express.Router()` 把路由按功能拆分到独立文件。

```js
// routes/users.js —— 用户相关的接口
const express = require('express');
const router = express.Router();

// GET /users —— 获取用户列表
router.get('/', (req, res) => {
  res.json([{ id: 1, name: 'Alice' }]);
});

// GET /users/:id —— 获取单个用户
router.get('/:id', (req, res) => {
  res.json({ id: req.params.id, name: 'Alice' });
});

// POST /users —— 创建用户
router.post('/', (req, res) => {
  // req.body 已经由 express.json() 解析好了
  const { name, email } = req.body;
  res.status(201).json({ id: Date.now(), name, email });
});

module.exports = router;
```

```js
// app.js —— 主入口，挂载路由模块
const express = require('express');
const app = express();

app.use(express.json());

// 所有以 /users 开头的请求，都交给 usersRouter 处理
const usersRouter = require('./routes/users');
app.use('/users', usersRouter);

// 其他模块同理
// const productsRouter = require('./routes/products');
// app.use('/products', productsRouter);

app.listen(3000, () => console.log('Server running on port 3000'));
```

**好处**：接口文件结构清晰，便于团队协作和维护。

### 2.4 CORS 跨域配置

**为什么需要？** 前端（`localhost:5173`）请求后端（`localhost:3000`）时，浏览器会因同源策略拦截。后端必须明确允许跨域。

```bash
npm install cors
```

```js
const cors = require('cors');

// 开发环境：允许所有来源跨域
app.use(cors());

// 生产环境：只允许指定域名
app.use(cors({
  origin: 'https://yourdomain.com',
  methods: ['GET', 'POST'],
  credentials: true  // 允许携带 Cookie
}));
```

放在所有路由前面，作为第一个中间件执行。

### 2.5 自定义中间件

中间件就是一个函数，能在请求到达路由之前做处理（鉴权、日志、参数校验等）。你可以自己写中间件插入到请求处理链中。

**示例：鉴权中间件**

```js
// 自定义鉴权中间件
function authMiddleware(req, res, next) {
  const token = req.headers.authorization;
  
  // 没有 token，直接返回 401，阻断后续流程
  if (!token) {
    return res.status(401).json({ error: '未登录，请先登录' });
  }
  
  // 验证 token（示意，实际需用 jsonwebtoken 等库）
  if (token !== 'secret-token') {
    return res.status(403).json({ error: 'token 无效' });
  }
  
  // 验证通过，把用户信息挂到 req 上，后面的路由可以直接用
  req.user = { id: 1, name: 'Alice', role: 'admin' };
  next(); // 放行，继续下一个中间件或路由
}

// 在需要保护的路由上使用
app.get('/profile', authMiddleware, (req, res) => {
  // req.user 是中间件挂上去的，前端无感
  res.json({ message: '个人中心', user: req.user });
});

app.get('/admin', authMiddleware, (req, res) => {
  if (req.user.role !== 'admin') {
    return res.status(403).send('无权限');
  }
  res.send('管理后台');
});
```

**中间件执行顺序**：

```
请求进来
  → cors()                    // 1. 跨域处理
  → express.json()            // 2. 解析 JSON 请求体
  → authMiddleware            // 3. 鉴权（如果匹配）
  → 路由处理函数              // 4. 业务逻辑
  → 响应返回
```

任何一个中间件调用 `res.send()` / `res.json()` 后，请求终止，不再往后传递。

---

### 注释（补充说明）

1. **`app.use(express.json())`**：解析 JSON 请求体，将字符串转成对象挂到 `req.body`。如果服务要接收 JSON 数据，这一行必不可少。

2. **`app.use` vs `app.all`**：
   - `app.use('/admin', ...)` ：匹配路径**前缀**，如 `/admin`、`/admin/users`、`/admin/settings`
   - `app.all('/admin', ...)` ：精确匹配 `/admin`，且**匹配所有 HTTP 方法**（GET、POST、PUT 等）

3. **`next()`**：调用 `next()` 才会将请求传递给下一个中间件或路由。如果某个中间件已经 `res.send()` 了，则不要再调用 `next()`，因为请求-响应周期已结束。

4. **路由参数 vs 查询参数**：
   - `req.params`：路径中的占位符，如 `/users/:id` → `req.params.id`
   - `req.query`：URL 问号后的键值对，如 `/users?name=john` → `req.query.name`

5. **CORS**：开发时务必配置，否则前端本地开发连不上后端。生产环境要限制来源。

**关键对比分析（更新）**：
```bash
| 功能 | 原生 Node.js | Express.js |
|------|-------------|-----------|
| 路由处理 | 手动解析 URL 和 method | 声明式路由 (`app.get()`, `app.post()`) |
| 查询参数 | 手动解析 `url.parse()` | 自动解析 (`req.query`) |
| 路径参数 | 需手动从 URL 中提取 | 占位符 `:id`，自动解析 (`req.params`) |
| 请求体解析 | 手动处理数据流 + JSON 解析 | 一行中间件 (`express.json()`) |
| 响应处理 | 手动设置 headers + `res.end()` | 智能方法 (`res.send()`, `res.json()`) |
| 404 处理 | 在每个分支手动判断 | 统一中间件处理 |
| 错误处理 | 需要 `try/catch` 包裹 | 统一错误处理中间件 |
| 路由拆分 | 完全手动 | `express.Router()` 模块化 |
| 跨域处理 | 手动设置 `Access-Control-*` 头 | `cors` 中间件一行搞定 |
| 中间件体系 | 无 | 灵活的自定义中间件链 |
```

## 3、服务请求+响应的完整路径
当然，express自然也是支持像fetch这样，发送服务请求的功能。
# (1) fetch
```js
fetch('https://api.example.com/users', {
  method: 'POST',
  headers: {
    'Content-Type': 'application/json',
    'Authorization': 'Bearer your_token' // 身份验证示例
  },
  body: JSON.stringify({ // 必须字符串化
    name: 'John',
    age: 30
  })
})
```
nodejs的http模块：https://www.nodeapp.cn/http.html#http_class_http_agent。总的来说，http 模块主要围绕两大核心功能展开：
1. http.createServer() - 创建 HTTP 服务器
2. http.request() - 发起 HTTP 客户端请求
其他类都是为这两大功能服务的"组件"或"工具类"。

# (2) express
```js
// 4. 创建服务请求
const clientReq = http.request('http://localhost:3000', (clientRes) => {
  // clientRes 是 IncomingMessage - 读取服务器响应
  let data = '';
  clientRes.on('data', chunk => data += chunk);
  clientRes.on('end', () => console.log('服务器响应:', data));
});

clientReq.end(); // 发送请求
```
